function formatMonth(m) {

  const [y, mo] = m.split('-');

  const n = ['Январь','Февраль','Март','Апрель','Май','Июнь','Июль','Август','Сентябрь','Октябрь','Ноябрь','Декабрь'];

  return n[parseInt(mo)-1] + ' ' + y;

}

function formatDate(d) { const [y,m,day] = d.split('-'); return day+'.'+m+'.'+y; }

function getMonths() { return [...new Set(state.events.map(e => e.date.slice(0,7)))].sort(); }

function getMonthEvents() { return state.events.filter(e => e.date.startsWith(currentMonth)).sort((a,b) => a.date.localeCompare(b.date) || a.title.localeCompare(b.title)); }

function escapeHtml(s) { return String(s||'').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

function round1(n) { return Math.round(n*10)/10; }

function uid(prefix) { return (prefix||'id') + '_' + Date.now() + '_' + Math.random().toString(36).slice(2,6); }


async function initData() {

  if (!isOnline) {

    try {

      const raw = localStorage.getItem('overtime_local_v4');

      if (raw) { state = JSON.parse(raw); return; }

    } catch(e) {}

    state = {

      employees: defaultEmployees.map(x => ({...x})),

      events: [...mayEvents.map(x => ({...x})), ...parseCompactEvents(compactEventsText)],

      hours: {}

    };

    saveLocal();

    return;

  }

  const metaSnap = await db.ref('meta/initialized').once('value');

  if (!metaSnap.val()) {

    const emps = {}; defaultEmployees.forEach(e => emps[e.id] = e);

    const evs = {};

    [...mayEvents, ...parseCompactEvents(compactEventsText)].forEach(e => evs[e.id] = e);

    await db.ref('employees').set(emps);

    await db.ref('events').set(evs);

    await db.ref('meta/initialized').set(true);

  }

}


function saveLocal() {

  try { localStorage.setItem('overtime_local_v4', JSON.stringify(state)); } catch(e) {}

}


function subscribeFirebase() {

  db.ref('employees').on('value', snap => {

    const val = snap.val() || {};

    state.employees = Object.values(val);

    if (!suppressRender) render();

  });

  db.ref('events').on('value', snap => {

    const val = snap.val() || {};

    state.events = Object.values(val);

    if (!suppressRender) render();

  });

  db.ref('hours').on('value', snap => {

    state.hours = snap.val() || {};

    updateHoursInPlace();

    updateSummary();

  });

  const uidKey = localStorage.getItem('presence_uid') || uid('u');

  localStorage.setItem('presence_uid', uidKey);

  const pRef = db.ref('presence/' + uidKey);

  pRef.onDisconnect().remove();

  pRef.set({ ts: Date.now() });

  setInterval(() => pRef.set({ ts: Date.now() }), 30000);

  db.ref('presence').on('value', snap => {

    const list = Object.values(snap.val() || {}).filter(p => Date.now() - p.ts < 120000);

    document.getElementById('onlineCount').style.display = list.length ? 'inline-flex' : 'none';

    document.getElementById('onlineNum').textContent = list.length;

  });

}

function render() {

  const months = getMonths();

  if (!months.length) {

    document.getElementById('monthTabs').innerHTML = '';

    document.getElementById('eventList').innerHTML = '';

    document.getElementById('mainPanel').innerHTML = '<div class="card"><div class="empty">Нет данных. Добавьте мероприятие.</div></div>';

    return;

  }

  if (!currentMonth || !months.includes(currentMonth)) {

    const today = new Date().toISOString().slice(0,7);

    currentMonth = months.includes(today) ? today : months[0];

  }

  const el = document.getElementById('monthTabs');

  el.innerHTML = months.map(m => '<button class="month-tab ' + (m===currentMonth?'active':'') + '" data-month="' + m + '">' + formatMonth(m) + '</button>').join('');

  el.querySelectorAll('.month-tab').forEach(b => b.onclick = () => { currentMonth = b.dataset.month; currentEventId = null; render(); });

  renderEventList();

  renderMainPanel();

}


function renderEventList() {

  const events = getMonthEvents();

  const el = document.getElementById('eventList');

  let html = '<div class="event-item summary-item ' + (currentEventId===null?'active':'') + '" data-id="__summary__">' +

    '<div class="event-date">Сводка</div><div class="event-title">📊 Итого за ' + formatMonth(currentMonth) + '</div></div>';

  if (!events.length) html += '<div class="empty">В этом месяце нет мероприятий</div>';

  else html += events.map(ev => '<div class="event-item ' + (ev.id===currentEventId?'active':'') + '" data-id="' + ev.id + '">' +

    '<div class="event-date">' + formatDate(ev.date) + '</div>' +

    '<div class="event-title">' + escapeHtml(ev.title) + '</div></div>').join('');

  el.innerHTML = html;

  el.querySelectorAll('.event-item').forEach(item => item.onclick = () => {

    currentEventId = item.dataset.id === '__summary__' ? null : item.dataset.id;

    renderEventList(); renderMainPanel();

  });

}


function getRoleFor(ev, empSurname) {

  const fields = [['responsible','Ответственный'],['sound','Звук'],['light','Свет'],['video','Видео'],['direction','Проведение']];

  for (const [f, label] of fields) if (ev[f] && ev[f].includes(empSurname)) return label;

  return '';

}


function renderMainPanel() {

  const el = document.getElementById('mainPanel');

  let html = '';

  if (currentEventId) {

    const ev = state.events.find(e => e.id === currentEventId);

    if (ev) html += renderEventCard(ev);

  }

  html += renderMonthlySummary();

  el.innerHTML = html;

  bindEventInputs();

}


function renderEventCard(ev) {

  const hours = state.hours[ev.id] || {};

  const meta = [];

  if (ev.location) meta.push('📍 ' + escapeHtml(ev.location));

  if (ev.producer) meta.push('🎬 ' + escapeHtml(ev.producer));

  if (ev.budgetCode) meta.push('💰 ' + escapeHtml(ev.budgetCode));

  const rows = state.employees.map(emp => {

    const role = getRoleFor(ev, emp.surname || emp.name.split(' ')[0]);

    const h = hours[emp.id] || '';

    return '<tr><td>' + escapeHtml(emp.name) + '</td>' +

      '<td class="position-cell">' + escapeHtml(emp.position || '') + '</td>' +

      '<td>' + (role ? '<span class="role-badge">' + role + '</span>' : '—') + '</td>' +

      '<td><input type="number" min="0" step="0.5" value="' + h + '" data-emp="' + emp.id + '" data-ev="' + ev.id + '" placeholder="0"></td></tr>';

  }).join('');

  return '<div class="card"><div class="toolbar" style="margin-bottom:8px;"><h2>' + escapeHtml(ev.title) + '</h2>' +

    '<div><button onclick="openEventDialog(\'' + ev.id + '\')">✎ Редактировать</button> <button onclick="deleteEvent(\'' + ev.id + '\')">🗑</button></div></div>' +

    '<div class="meta"><span>📅 ' + formatDate(ev.date) + '</span>' + meta.map(m => '<span>' + m + '</span>').join('') + '</div>' +

    '<table><thead><tr><th>Сотрудник</th><th>Должность</th><th>Роль</th><th>Часы</th></tr></thead><tbody>' + rows + '</tbody></table>' +

    '<div class="event-total-bar">Итого за мероприятие: <strong id="eventTotal">' + calcEventTotal(ev.id) + ' ч</strong></div></div>';

}

function calcEventTotal(evId) { const h = state.hours[evId] || {}; return round1(Object.values(h).reduce((s,v) => s + (Number(v)||0), 0)); }


function bindEventInputs() {

  document.querySelectorAll('input[data-emp]').forEach(inp => {

    inp.oninput = () => {

      const evId = inp.dataset.ev, empId = inp.dataset.emp, val = parseFloat(inp.value);

      if (!state.hours[evId]) state.hours[evId] = {};

      if (val && val > 0) state.hours[evId][empId] = val; else delete state.hours[evId][empId];

      if (isOnline) {

        db.ref('hours/' + evId + '/' + empId).set(val && val > 0 ? val : null);

      } else {

        saveLocal();

        const tot = document.getElementById('eventTotal');

        if (tot) tot.textContent = calcEventTotal(evId) + ' ч';

        updateSummary();

      }

    };

  });

}


function updateHoursInPlace() {

  document.querySelectorAll('input[data-emp]').forEach(inp => {

    if (document.activeElement === inp) return;

    const evId = inp.dataset.ev, empId = inp.dataset.emp;

    const val = state.hours[evId]?.[empId] || '';

    if (String(inp.value) !== String(val)) inp.value = val;

  });

  const tot = document.getElementById('eventTotal');

  if (tot && currentEventId) tot.textContent = calcEventTotal(currentEventId) + ' ч';

}


function renderMonthlySummary() {

  const events = getMonthEvents();

  const dates = [...new Set(events.map(e => e.date))].sort();

  const rows = state.employees.map(emp => {

    let totalHours = 0, overtime = 0, workDays = 0;

    for (const date of dates) {

      let dayTotal = 0;

      for (const ev of events) if (ev.date === date) dayTotal += (Number(state.hours[ev.id]?.[emp.id]) || 0);

      if (dayTotal > 0) workDays++;

      totalHours += dayTotal;

      if (dayTotal > 8) overtime += dayTotal - 8;

    }

    return { name: emp.name, position: emp.position || '', totalHours: round1(totalHours), workDays, overtime: round1(overtime) };

  });

  const totalAll = round1(rows.reduce((s,r) => s + r.totalHours, 0));

  const totalOt = round1(rows.reduce((s,r) => s + r.overtime, 0));

  return '<div class="card" id="summary"><h2>Итого за ' + formatMonth(currentMonth) + '</h2>' +

    '<p class="hint">Норма — 8 часов в день. Переработка = часы сверх 8 за день (суммируются все мероприятия дня).</p>' +

    '<table><thead><tr><th>Сотрудник</th><th>Должность</th><th>Рабочих дней</th><th>Всего часов</th><th>Переработка</th></tr></thead><tbody>' +

    rows.map(r => '<tr><td>' + escapeHtml(r.name) + '</td><td class="position-cell">' + escapeHtml(r.position) + '</td>' +

      '<td>' + (r.workDays||'—') + '</td><td>' + (r.totalHours||'—') + '</td>' +

      '<td class="' + (r.overtime>0?'overtime':'') + '">' + (r.overtime>0?r.overtime:'—') + '</td></tr>').join('') +

    '</tbody><tfoot><tr><td>Итого</td><td></td><td></td><td>' + totalAll + '</td><td class="' + (totalOt>0?'overtime':'') + '">' + (totalOt||'—') + '</td></tr></tfoot></table></div>';

}


function updateSummary() {

  const old = document.getElementById('summary');

  if (!old) return;

  const tmp = document.createElement('div');

  tmp.innerHTML = renderMonthlySummary();

  old.replaceWith(tmp.firstElementChild);

}

function addEvent() {

  const dlg = document.getElementById('eventDialog');

  const date = document.getElementById('newEventDate').value;

  const title = document.getElementById('newEventTitle').value.trim();

  if (!date || !title) { alert('Укажите дату и название'); return; }

  const data = {

    date, title,

    location: document.getElementById('newEventLocation').value.trim(),

    producer: document.getElementById('newEventProducer').value.trim(),

    responsible: document.getElementById('newEventResponsible').value.trim(),

    sound: document.getElementById('newEventSound').value.trim(),

    light: document.getElementById('newEventLight').value.trim(),

    video: document.getElementById('newEventVideo').value.trim(),

    direction: document.getElementById('newEventDirection').value.trim(),

    budgetCode: document.getElementById('newEventBudget').value.trim(),

  };

  if (dlg.dataset.editId) {

    const id = dlg.dataset.editId;

    const ev = state.events.find(e => e.id === id);

    if (ev) Object.assign(ev, data);

    if (isOnline) db.ref('events/' + id).set({ ...ev, ...data });

    else { saveLocal(); render(); }

  } else {

    const id = uid('ev');

    const ev = { id, ...data };

    if (isOnline) db.ref('events/' + id).set(ev);

    else { state.events.push(ev); saveLocal(); }

    currentMonth = date.slice(0,7); currentEventId = null;

    if (!isOnline) render();

  }

  dlg.close();

}

function deleteEvent(id) {

  if (!confirm('Удалить мероприятие и все введённые часы?')) return;

  if (isOnline) { db.ref('events/' + id).remove(); db.ref('hours/' + id).remove(); }

  else { state.events = state.events.filter(e => e.id !== id); delete state.hours[id]; saveLocal(); render(); }

  if (currentEventId === id) currentEventId = null;

}


function exportCSV() {

  const events = getMonthEvents();

  const dates = [...new Set(events.map(e => e.date))].sort();

  const lines = ['Сотрудник;Должность;Рабочих дней;Всего часов;Переработка'];

  for (const emp of state.employees) {

    let total = 0, ot = 0, days = 0;

    for (const date of dates) {

      let d = 0; for (const ev of events) if (ev.date === date) d += (Number(state.hours[ev.id]?.[emp.id])||0);

      if (d > 0) days++; total += d; if (d > 8) ot += d - 8;

    }

    if (total > 0) lines.push([emp.name, emp.position||'', days, round1(total), round1(ot)].join(';'));

  }

  download('\uFEFF' + lines.join('\n'), 'pererabotki_' + currentMonth + '.csv', 'text/csv;charset=utf-8');

}

function download(content, filename, mime) {

  const blob = new Blob([content], { type: mime });

  const url = URL.createObjectURL(blob);

  const a = document.createElement('a'); a.href = url; a.download = filename; a.click();

  setTimeout(() => URL.revokeObjectURL(url), 1000);

}
